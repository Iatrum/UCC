import { test, expect, Page } from '@playwright/test';
import fs from 'node:fs';
import { watch, shot, uiLogin, saveJson, Issue } from '../lib/app';
import { byPatient } from '../lib/fhir';
test.use({ trace: 'off' });
const facts: Record<string, any> = {}; const issues: Issue[] = [];
test('mc reissue + documents tab letterhead', async ({ browser }) => {
  test.setTimeout(15 * 60_000);
  const patientId = JSON.parse(fs.readFileSync('/tmp/ev605/walk605-results.json', 'utf8')).facts.patientId;
  const ctx = await browser.newContext({ storageState: { cookies: [], origins: [] } });
  const page: Page = await ctx.newPage(); watch(page, issues, () => 'b');
  await uiLogin(page);
  await page.goto(`/patients/${patientId}`); await page.getByRole('tab', { name: 'Treatment', exact: true }).click();
  await page.locator('input:visible').last().fill('medical');
  const b = page.locator('button', { hasText: /medical certificate/i }).filter({ hasText: 'RM' }).first(); await expect(b).toBeVisible(); await b.click(); await page.waitForTimeout(1200);
  await page.getByText(/MEDICAL CERTIFICATE/).last().click();
  const dlg = page.getByRole('dialog'); await expect(dlg).toBeVisible();
  const f = (l: string) => dlg.locator(`xpath=.//label[normalize-space()="${l}"]/following-sibling::input[1]`);
  await f('No of days').fill('2'); await f("Doctor's name").fill('Dr Fake Walk'); await f('MMC registration no.').fill('MMC-FAKE-605B');
  await f('Diagnosis').fill('Acute pharyngitis'); 
  let txt = await dlg.frameLocator('iframe[title="MC preview"]').locator('body').innerText(); facts.mcNoDx = txt.replace(/\s+/g, ' ');
  expect(txt).not.toContain('Acute pharyngitis'); expect(txt).toContain('(2 days)'); expect(txt).toContain('MMC-FAKE-605B');
  await dlg.getByLabel(/Show diagnosis on the MC/).check();
  txt = await dlg.frameLocator('iframe[title="MC preview"]').locator('body').innerText(); facts.mcDx = txt.replace(/\s+/g, ' ');
  expect(txt).toContain('Acute pharyngitis');
  await dlg.getByRole('button', { name: 'Complete document' }).click(); await expect(dlg).toHaveCount(0);
  await page.getByRole('button', { name: 'Sign', exact: true }).click();
  await expect.poll(async () => (await byPatient('Procedure', patientId)).filter((p: any) => /MEDICAL CERT/i.test(p.code?.text || '')).length, { timeout: 90_000 }).toBeGreaterThanOrEqual(2);
  // Documents tab: wait for the org fetch, then open
  const org = page.waitForResponse(r => r.url().includes('/api/organization'));
  await page.goto(`/patients/${patientId}`); await page.getByRole('tab', { name: 'Documents' }).click(); await org; await page.waitForTimeout(500);
  const rows = page.getByRole('button', { name: /^(Medical Certificate|MC)/i }); facts.mcRows = await rows.count();
  for (let i = 0; i < facts.mcRows; i++) {
    await rows.nth(i).click(); const t = (await page.frameLocator('iframe[title="Document preview"]').locator('body').innerText()).replace(/\s+/g, ' ');
    facts[`mcDoc${i}`] = t.slice(0, 450); await shot(page, `b-mc-${i}`); await page.getByRole('button', { name: 'Close' }).last().click(); await page.waitForTimeout(400);
  }
  // Documents tab opened immediately after load (no wait for org)
  await page.goto(`/patients/${patientId}`); await page.getByRole('tab', { name: 'Documents' }).click();
  await page.getByRole('button', { name: /^Time Chit/ }).first().click();
  facts.fastOpenTimeChit = (await page.frameLocator('iframe[title="Document preview"]').locator('body').innerText()).replace(/\s+/g, ' ').slice(0, 120);
  await page.getByRole('button', { name: 'Close' }).last().click();
  saveJson('walk605b.json', { facts, issues });
  await page.reload(); await page.waitForTimeout(1500);
  await page.getByRole('tab', { name: 'Consultation History' }).click(); await page.getByRole('button', { name: 'View' }).first().click({ timeout: 15000 }).catch(e => { facts.viewErr = String(e).slice(0, 200); }); await page.waitForTimeout(2500);
  facts.viewText = (await page.locator('body').innerText()).replace(/\s+/g, ' ').match(/.{0,200}(time chit|medical certificate).{0,200}/i)?.[0] || 'NO DOC MENTION'; await shot(page, 'b-history-view');
  saveJson('walk605b.json', { facts, issues });
});
