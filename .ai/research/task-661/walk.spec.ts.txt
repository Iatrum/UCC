// Task 661 policied pilot walk (A12): same walk as Task 589, but as a staff user whose membership carries the clinic AccessPolicy
// (non-superAdmin project), plus cross-clinic read/write denial probes. Originally Task 589 pilot regression walk. One page, fake data. Every step is recorded pass/fail and the walk continues past failures.
test.use({ trace: 'off' });
const issues: Issue[] = []; const results: { step: string; ok: boolean; note?: string }[] = []; const facts: Record<string, any> = { run: RUN };
const name = `Policied Walk Fake ${RUN}`;
const nric = `9203${String(1 + Math.floor(Math.random() * 28)).padStart(2, '0')}-14-${String(Math.floor(Math.random() * 10000)).padStart(4, '0')}`;
let cur = 'init';

test('pilot walk', async ({ browser }) => {
  test.setTimeout(30 * 60_000);
  const ctx = await browser.newContext({ storageState: { cookies: [], origins: [] } });
  const page: Page = await ctx.newPage(); watch(page, issues, () => cur);
  const S = async (id: string, fn: () => Promise<void>) => {
    cur = id;
    try { await fn(); results.push({ step: id, ok: true }); await shot(page, id); }
    catch (e: any) { results.push({ step: id, ok: false, note: String(e.message).split('\n').slice(0, 4).join(' | ').slice(0, 500) }); await shot(page, `FAIL-${id}`); }
  };
  const pick = async (q: string, re: RegExp) => {
    await page.locator('input:visible').last().fill(q);
    const btn = page.locator('button', { hasText: re }).filter({ hasText: 'RM' }).first();
    await expect(btn).toBeVisible({ timeout: 30_000 }); await btn.click(); await page.waitForTimeout(1200);
  };
  await S('01-login', async () => { await uiLogin(page); });
  await S('02-register', async () => {
    await page.goto('/patients/new');
    await expect(async () => { await page.fill('[name=fullName]', name); await page.fill('[name=nric]', nric); await expect(page.locator('[name=nric]')).toHaveValue(nric); }).toPass({ timeout: 30_000 });
    await page.locator('[name=nric]').blur(); await page.selectOption('select', 'male').catch(() => {});
    await page.fill('[name=email]', `walk.${RUN}@example.test`); await page.fill('[name=phone]', '0123456789');
    await page.fill('[name=address]', '1 Jalan Ujian, Kuala Lumpur'); await page.fill('[name=postalCode]', '50000');
    await page.fill('[name="emergencyContact.name"]', 'Aminah Fake'); await page.fill('[name="emergencyContact.relationship"]', 'Spouse');
    await page.fill('[name="emergencyContact.phone"]', '0198765432'); await page.fill('[name="medicalHistory.allergies"]', 'Penicillin');
    await page.getByRole('button', { name: 'Register Patient' }).click();
    await page.waitForURL(/\/patients\/[0-9a-f-]{36}\/check-in/, { timeout: 120_000 });
    facts.patientId = page.url().match(/patients\/([0-9a-f-]{36})/)![1];
  });
  await S('03-checkin-triage', async () => {
    await page.fill('#chief-complaint', 'Fever and sore throat x3 days (walk fake)');
    const vals = ['128', '82', '96', '18', '38.4', '98', '3', '70', '170']; const nums = page.locator('input[type=number]');
    for (let i = 0; i < Math.min(vals.length, await nums.count()); i++) await nums.nth(i).fill(vals[i]);
    await page.getByRole('button', { name: /Complete check-in/ }).click(); await page.waitForURL(/dashboard/, { timeout: 120_000 });
    await expect(page.locator('tr', { hasText: name })).toBeVisible({ timeout: 60_000 });
    expect((await byPatient('Observation', facts.patientId)).length).toBeGreaterThanOrEqual(9);
  });
  await S('04-start-consult-notes-diagnosis-sign', async () => {
    await page.locator('tr', { hasText: name }).locator('button').last().click();
    await page.getByRole('menuitem', { name: /Start Consultation/i }).click();
    await page.waitForURL(new RegExp(facts.patientId), { timeout: 120_000 });
    await page.getByRole('button', { name: 'Start consult' }).click();
    await page.locator('[contenteditable=true]').first().click();
    await page.keyboard.type('S: Fever, sore throat 3/7. O: T 38.4. A: Acute pharyngitis. P: symptomatic, MC 2 days. (walk fake)');
    await page.locator('button', { hasText: 'Select or search diagnosis' }).click(); await page.keyboard.type('pharyng');
    await page.locator('[role=option]', { hasText: 'J02.9' }).first().dispatchEvent('click');
    await expect(page.locator('button', { hasText: 'Acute pharyngitis' })).toBeVisible();
    await shot(page, '04a-consult-filled');
    await page.getByRole('button', { name: 'Sign', exact: true }).click();
    await expect(page.getByText('Order', { exact: true })).toBeVisible({ timeout: 90_000 });
    expect((await byPatient('Condition', facts.patientId)).length).toBeGreaterThan(0);
  });
  await S('05-composer-add-med-lab-mc-timechit', async () => {
    // Real autosave path first (Firestore may be unconfigured locally); record what happens, stub only if it rolls back.
    const planResp = page.waitForResponse(r => r.url().includes('/api/consultations/plan') && r.request().method() === 'POST', { timeout: 60_000 }).catch(() => null);
    await pick('para', /Paracetamol/);
    const pr = await planResp; facts.planPostStatus = pr?.status(); await page.waitForTimeout(2500);
    facts.itemKeptRealPath = (await page.getByText(/1 × RM 0\.50/).count()) > 0;
    if (!facts.itemKeptRealPath) {
      await page.route('**/api/consultations/plan**', r => r.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ success: true, persistenceAvailable: false, plan: { entries: [], summary: {} } }) }));
      await page.reload(); await page.getByRole('tab', { name: 'Treatment' }).click(); await page.waitForTimeout(2500); await page.keyboard.press('Escape');
      await pick('para', /Paracetamol/);
    }
    await pick('full blood', /Full Blood/);
    await pick('medical', /medical certificate/i);
    await page.getByText(/MEDICAL CERTIFICATE/).last().click();
    let dlg = page.getByRole('dialog'); await expect(dlg).toBeVisible();
    await dlg.getByPlaceholder('Dr. ...').fill('Dr Fake Walk'); await shot(page, '05a-mc-dialog');
    await dlg.getByRole('button', { name: 'Complete document' }).click(); await expect(dlg).toHaveCount(0);
    await page.locator('input:visible').last().fill('time chit');
    const tc = page.getByRole('button', { name: /TIME CHIT.*RM/i }); await expect(tc).toBeVisible({ timeout: 30_000 }); await tc.click();
    await page.getByText('TIME CHIT', { exact: true }).last().click();
    dlg = page.getByRole('dialog'); await expect(dlg).toBeVisible();
    await dlg.getByLabel('Attendance date').fill('2026-11-18'); await dlg.getByLabel('Time in', { exact: true }).fill('09:00');
    await dlg.getByLabel('Time out', { exact: true }).fill('10:30'); await dlg.getByLabel("Doctor's name").fill('Dr Fake Walk');
    await expect(dlg.getByText('Duration: 1.5 hour(s)')).toBeVisible(); await shot(page, '05b-timechit-dialog');
    await dlg.getByRole('button', { name: 'Complete document' }).click(); await expect(dlg).toHaveCount(0);
    await page.locator('input:visible').last().fill(''); await shot(page, '05c-composer-filled');
    await expect(page.getByText('Paracetamol').last()).toBeVisible();
    // composer must not show critical-result warnings
    expect(await page.getByText(/critical/i).count()).toBe(0);
  });
  await S('06-sign-treatment', async () => {
    await page.getByRole('button', { name: 'Sign', exact: true }).click();
    await expect.poll(async () => { const [m, o] = await Promise.all([byPatient('MedicationRequest', facts.patientId), byPatient('ServiceRequest', facts.patientId)]); return m.length > 0 && o.length >= 1; }, { timeout: 90_000 }).toBe(true);
    const proc = await byPatient('Procedure', facts.patientId);
    facts.procedures = proc.map((p: any) => `${p.code?.text}:${p.status}`);
    expect(proc.some((p: any) => /TIME CHIT/i.test(p.code?.text || ''))).toBe(true);
    expect(proc.some((p: any) => /MEDICAL CERT/i.test(p.code?.text || ''))).toBe(true);
    facts.labSR = (await byPatient('ServiceRequest', facts.patientId)).find((s: any) => /blood/i.test(s.code?.text || ''))?.id;
  });
  await S('07-lab-ingest-then-labs-tab', async () => {
    expect(facts.labSR, 'FBC ServiceRequest id').toBeTruthy();
    const m = await fhir();
    const clinicId = (await m.readResource('Patient', facts.patientId)).identifier?.find((id: any) => id.system === 'clinic')?.value as string;
    facts.walkClinic = clinicId;
    const restorePoct = await setClinicModule(m, clinicId, 'poct', true);
    const { key: deviceKey, identifier: keyHash } = newDeviceKey();
    const analyser = await m.createResource<any>({ resourceType: 'Device', status: 'active', identifier: [{ system: 'clinic', value: clinicId }, keyHash],
      meta: { tag: [{ system: 'https://ucc.emr/poct', code: 'simulated' }] }, deviceName: [{ name: `Policied Walk Fake analyser ${RUN}`, type: 'user-friendly-name' }],
      type: { coding: [{ system: 'https://ucc.emr/poct/analyser-type', code: 'glucose' }] } });
    // Warm the cold route compile (capped dev server) before the real ingest.
    await page.request.post('/api/labs/receive', { data: {}, timeout: 180_000 }).catch(() => {});
    try {
      const r = await page.request.post('/api/labs/receive', { timeout: 180_000, headers: bearer(deviceKey), data: { serviceRequestId: facts.labSR, conclusion: 'Mild leukocytosis (FAKE)',
        results: [{ testCode: '718-7', testName: 'Haemoglobin', value: 13.9, unit: 'g/dL', referenceRange: '13-17', status: 'final', interpretation: 'normal' },
                  { testCode: '6690-2', testName: 'White cell count', value: 12.4, unit: '10*9/L', referenceRange: '4-11', status: 'final', interpretation: 'high' }] } });
      facts.labReceiveStatus = r.status(); expect(r.ok()).toBeTruthy();
    } finally { await m.deleteResource('Device', analyser.id!); await restorePoct(); }
    await page.unroute('**/api/consultations/plan**').catch(() => {});
    await page.goto(`/patients/${facts.patientId}`);
    const t = page.getByRole('tab', { name: 'Labs & Imaging', exact: true });
    await expect(async () => { await t.click(); await expect(t).toHaveAttribute('aria-selected', 'true', { timeout: 2000 }); }).toPass({ timeout: 30_000 });
    await expect(page.getByText(/Haemoglobin|White cell|Full Blood/).first()).toBeVisible({ timeout: 60_000 });
    facts.labsTabShowsResult = true;
  });
  await S('08-documents-tab-mc-timechit', async () => {
    await page.goto(`/patients/${facts.patientId}`); await page.getByRole('tab', { name: 'Documents' }).click();
    await expect(page.getByRole('button', { name: /^Time Chit/ }).first()).toBeVisible({ timeout: 30_000 });
    await page.getByRole('button', { name: /^Time Chit/ }).first().click();
    await expect(page.frameLocator('iframe[title="Document preview"]').getByText('TIME CHIT / SURAT MASA')).toBeVisible({ timeout: 30_000 });
    await shot(page, '08a-timechit-preview'); await page.keyboard.press('Escape');
    const mc = page.getByRole('button', { name: /^(Medical Certificate|MC)/i }).first();
    if (await mc.count()) { await mc.click(); await page.waitForTimeout(2500); await shot(page, '08b-mc-preview'); }
    else throw new Error('no MC button in Documents tab');
  });
  await S('09-checkout-dispense-invoice', async () => {
    const before: any = (await (await page.request.get('/api/inventory')).json()).medications?.find((x: any) => x.name === 'Paracetamol')?.stock;
    facts.stockBefore = before;
    await page.goto('/orders'); await page.getByRole('link', { name }).first().click();
    await page.waitForURL(/orders\/checkout\//, { timeout: 120_000 });
    await expect(page.getByText('Loading checkout...', { exact: true })).toHaveCount(0); await expect(page.getByText('Payments', { exact: true })).toBeVisible();
    await shot(page, '09a-checkout');
    const body = await page.locator('main').last().innerText(); const m = body.match(/Total[^\d]*RM\s*([\d.,]+)/i); expect(m).toBeTruthy();
    facts.total = m![1]; await page.locator('#paid-amount').fill(m![1].replace(/,/g, ''));
    await page.getByRole('button', { name: /Complete visitation/ }).click(); await page.getByRole('button', { name: 'Complete', exact: true }).click();
    await page.waitForURL(/checkout=completed/, { timeout: 120_000 });
    const mp = await fhir();
    const inv = await mp.searchResources('Invoice', { subject: `Patient/${facts.patientId}` }, { cache: 'no-cache' }); expect(inv.length).toBeGreaterThan(0);
    facts.invoice = inv.map((i: any) => `${i.status}:${i.totalGross?.value}`);
    const after: any = (await (await page.request.get('/api/inventory')).json()).medications?.find((x: any) => x.name === 'Paracetamol')?.stock;
    facts.stockAfter = after; expect(after).toBe(before - 1);
    let n = 0; for (const enc of await byPatient('Encounter', facts.patientId)) n += ((await (await page.request.get(`/api/inventory?dispensesFor=${enc.id}`)).json()).dispenses || []).length;
    facts.dispenses = n; expect(n).toBe(1);
  });
  // ---- Enforcement: same policied staff must not read/write a second clinic's patient (Medplum AccessPolicy, not app checks) ----
  const base = MEDPLUM.replace(/\/$/, '');
  const raw = (token: string, p: string, method = 'GET', body?: any) => fetch(`${base}/fhir/R4/${p}`, { method, redirect: 'error', signal: AbortSignal.timeout(30_000),
    headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/fhir+json', 'X-Medplum': 'extended' }, body: body ? JSON.stringify(body) : undefined });
  const staffToken = async (email: string) => {
    await new Promise(r => setTimeout(r, 15_000)); // Medplum auth budget is 5/min
    const verifier = randomBytes(32).toString('base64url');
    const c = new MedplumClient({ baseUrl: MEDPLUM, fetch: globalThis.fetch.bind(globalThis) });
    const lr = await c.startLogin({ email, password: PASSWORD, scope: 'openid profile offline_access', codeChallenge: createHash('sha256').update(verifier).digest('base64url'), codeChallengeMethod: 'S256' });
    const tr = await fetch(new URL('oauth2/token', MEDPLUM), { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({ grant_type: 'authorization_code', code: lr.code!, code_verifier: verifier }) });
    expect(tr.status, 'staff token exchange').toBe(200);
    return (await tr.json()).access_token as string;
  };
  const owned: Array<{ resourceType: any; id: string }> = [];
  const denied = (s: number) => expect([403, 404], `status ${s}`).toContain(s);
  const probes: Record<string, any> = {};
  let foreign: any;
  await S('14-fixture-foreign-clinic-patient', async () => {
    const m = await fhir();
    const slug = 'clinicb';
    const account = await m.searchOne('HealthcareService', { identifier: `clinic|${slug}` }, { cache: 'no-cache' });
    expect(account?.id).toBeTruthy();
    const identifier = [{ system: 'clinic', value: slug }];
    const mk = async (r: any) => { const s = await m.createResource<any>(r); owned.push({ resourceType: s.resourceType, id: s.id! }); return s; };
    const patient = await mk({ resourceType: 'Patient', identifier, name: [{ family: `Policied Walk Foreign ${RUN}` }] });
    await m.post(`fhir/R4/Patient/${patient.id}/$set-accounts`, { resourceType: 'Parameters', parameter: [{ name: 'accounts', valueReference: { reference: `HealthcareService/${account!.id}` } }, { name: 'propagate', valueBoolean: false }] });
    const subject = { reference: `Patient/${patient.id}` };
    const encounter = await mk({ resourceType: 'Encounter', identifier, subject, status: 'in-progress', class: { system: 'http://terminology.hl7.org/CodeSystem/v3-ActCode', code: 'AMB' } });
    await m.post(`fhir/R4/Encounter/${encounter.id}/$set-accounts`, { resourceType: 'Parameters', parameter: [{ name: 'accounts', valueReference: { reference: `HealthcareService/${account!.id}` } }, { name: 'propagate', valueBoolean: false }] });
    foreign = { patient, encounter, account };
  });
  try {
    await S('15-staff-demo-own-clinic-allowed-foreign-denied', async () => {
      expect(foreign, 'foreign fixture').toBeTruthy(); expect(USER).toMatch(/demo/);
      const t = await staffToken(USER);
      const own = await raw(t, `Patient/${facts.patientId}`); expect(own.status, 'positive control: own-clinic read').toBe(200);
      const ownBody = await own.json();
      probes.ownRead = 200;
      probes.ownUpdate = (await raw(t, `Patient/${facts.patientId}`, 'PUT', ownBody)).status; expect(probes.ownUpdate, 'positive control: own-clinic write').toBe(200);
      const fp = foreign.patient.id;
      probes.foreignRead = (await raw(t, `Patient/${fp}`)).status; denied(probes.foreignRead);
      const s1 = await raw(t, `Patient?_id=${fp}`); probes.foreignSearchById = (await s1.json()).entry?.length || 0; expect(probes.foreignSearchById).toBe(0);
      const all = await (await raw(t, 'Patient?_count=200')).json(); const ents = all.entry || [];
      probes.patientSearchTotal = ents.length; expect(ents.length).toBeGreaterThan(0);
      const foreignHits = ents.filter((e: any) => !(e.resource.identifier || []).some((i: any) => i.system === 'clinic' && i.value === 'demo'));
      probes.patientSearchForeignClinicHits = foreignHits.length; expect(foreignHits.length).toBe(0);
      const encSearch = await (await raw(t, `Encounter?patient=${fp}`)).json(); expect(encSearch.entry?.length || 0).toBe(0);
      const fresh = await (await (await import('../lib/fhir')).fhir()).readResource('Patient', fp, { cache: 'no-cache', headers: { 'X-Medplum': 'extended' } });
      probes.foreignUpdate = (await raw(t, `Patient/${fp}`, 'PUT', { ...fresh, name: [{ family: 'TAMPERED' }] })).status; denied(probes.foreignUpdate);
      probes.foreignDelete = (await raw(t, `Patient/${fp}`, 'DELETE')).status; denied(probes.foreignDelete);
      const sr = await raw(t, 'ServiceRequest', 'POST', { resourceType: 'ServiceRequest', status: 'active', intent: 'order', subject: { reference: `Patient/${fp}` }, identifier: [{ system: 'clinic', value: 'clinicb' }], code: { text: 'forged foreign order' } });
      probes.foreignOrderCreate = sr.status; if (sr.ok) { const s = await sr.json(); owned.push({ resourceType: 'ServiceRequest', id: s.id }); } expect(sr.status).toBe(403);
      const fp2 = await raw(t, 'Patient', 'POST', { resourceType: 'Patient', identifier: [{ system: 'clinic', value: 'clinicb' }], meta: { accounts: [{ reference: `HealthcareService/${foreign.account.id}` }] }, name: [{ family: 'forged' }] });
      probes.forgedForeignPatientCreate = fp2.status; if (fp2.ok) { const s = await fp2.json(); owned.push({ resourceType: 'Patient', id: s.id }); } expect(fp2.status).toBe(403);
      // Raw project-level reads the policy must not grant
      for (const type of ['ProjectMembership', 'User', 'AccessPolicy']) { const r = await raw(t, `${type}?_count=1`); probes[`list${type}`] = r.status === 200 ? ((await r.json()).entry?.length || 0) : r.status; }
    });
    await S('16-staff-clinicb-cannot-touch-walk-patient', async () => {
      const t = await staffToken('task5-clinicb@example.invalid');
      const pid = facts.patientId;
      probes.bRead = (await raw(t, `Patient/${pid}`)).status; denied(probes.bRead);
      probes.bSearch = ((await (await raw(t, `Patient?_id=${pid}`)).json()).entry?.length || 0); expect(probes.bSearch).toBe(0);
      const fresh = await (await fhir()).readResource('Patient', pid, { cache: 'no-cache', headers: { 'X-Medplum': 'extended' } });
      probes.bUpdate = (await raw(t, `Patient/${pid}`, 'PUT', { ...fresh, name: [{ family: 'TAMPERED' }] })).status; denied(probes.bUpdate);
      const sr = await raw(t, 'ServiceRequest', 'POST', { resourceType: 'ServiceRequest', status: 'active', intent: 'order', subject: { reference: `Patient/${pid}` }, identifier: [{ system: 'clinic', value: 'demo' }], code: { text: 'forged foreign order' } });
      probes.bOrderCreate = sr.status; if (sr.ok) { const s = await sr.json(); owned.push({ resourceType: 'ServiceRequest', id: s.id }); } expect(sr.status).toBe(403);
      expect((await fresh).name?.[0]?.family).not.toBe('TAMPERED');
    });
    await S('17-app-layer-foreign-clinic-denied', async () => {
      // Same demo staff session through the app's own API (second layer): foreign patient/encounter yield nothing.
      const inv = await page.request.get(`/api/billing?patientId=${foreign.patient.id}`); probes.appBillingForeignPatient = inv.status();
      const j = await inv.json().catch(() => ({})); expect((j.invoices || []).length).toBe(0);
      const byEnc = await page.request.get(`/api/billing?consultationId=${foreign.encounter.id}`); probes.appBillingForeignEncounter = byEnc.status(); expect(byEnc.status()).not.toBe(200);
    });
  } finally {
    const m = await fhir();
    for (const o of owned.reverse()) await m.deleteResource(o.resourceType, o.id).catch(() => {});
  }
  facts.enforcementProbes = probes;
  facts.results = results;
  saveJson('walk-results.json', { facts, results, issues });
  const failed = results.filter(r => !r.ok); expect.soft(failed, JSON.stringify(failed)).toHaveLength(0);
  await ctx.close();
});
